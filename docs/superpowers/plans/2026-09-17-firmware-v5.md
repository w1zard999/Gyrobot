# Прошивка v5 (двухконтурный балансир) — план реализации

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** заменить прошивку гиробота на новую: каскад «PD угла (гироскоп, 200 Гц) + PI скорости (энкодеры, 50 Гц) + P позиции», с удержанием места и возвратом после толчка.

**Architecture:** один скетч из шести файлов (main/config/imu/motors/control/protocol). Внутренний контур угла по комплементарному фильтру, внешний контур скорости смещает уставку угла (±tiltmax), позиция задаёт уставку скорости. Протокол Serial — надмножество старого.

**Tech Stack:** Arduino Nano (ATmega328P), Wire, EEPROM, arduino-cli (avr core), avrdude, Python 3.14 + pyserial (tools).

**Спека:** `docs/specs/2026-09-17-firmware-v5-design.md` — формулы, значения по умолчанию, критерии испытаний.

## Global Constraints

- Плата: arduino:avr:nano (ATmega328P, 2 КБ RAM). Ориентир RAM после компиляции ≤ 75%.
- Библиотеки только Wire/EEPROM/Serial (входит в ядро). Никаких PID-библиотек.
- Пины строго из STATUS.md: PWMA=9 AIN1=7 AIN2=8 PWMB=10 BIN1=5 BIN2=4 STBY=6, энкодеры A: 2/A2, B: 3/A3, I2C: A4/A5.
- Дефолты смещений = живые значения 2026-09-17: gbx=319 gby=46 gbz=33 abx=234 aby=−14808 abz=426, оси axu=3 axf=2 axg=1 axt=3.
- `autoen=0` в дефолтах первой прошивки (моторы не оживут до проверки знаков).
- Все git-команды показывать пользователю с выводом.
- Каждый таск завершается коммитом.

### Task 3: imu.ino — MPU6050 + фильтр

**Files:** Create: `firmware/gyrobot_v5/imu.ino`

**Interfaces:** Consumes: cfg (смещения, оси, alpha). Produces: `bool imuInit(); bool imuRead(); float imuAngle; float imuGyro; uint16_t imuErr;` `void gyroCalib();` (команда c), `void rawCmd();`

- [ ] **Шаг 1: код**

```cpp
// MPU6050: ±500°/с, DLPF=3, смещения из cfg, комплементарный фильтр.
// Ось: axf — «вперёд», axu — «вверх»; угол = atan2(fwd, up), 0 = стойка,
// растёт при наклоне вперёд (STATUS.md). Тilt-гироскоп = ось axg, знак axg.
float imuAngle = 0, imuGyro = 0;  // угол °, скорость °/с
uint16_t imuErr = 0;
static float bootA = 0; static uint16_t bootN = 0;

static int16_t ax_, ay_, az_, gx_, gy_, gz_;

static float axisPick(float axx, int8_t sel) {  // sel: ±1/±2/±3
  float v = 0, s = (sel < 0) ? -1.0f : 1.0f;
  switch (sel < 0 ? -sel : sel) {
    case 1: v = ax_; break;  case 2: v = ay_; break;  default: v = az_;
  }
  return s * v;
}

bool imuInit() {
  Wire.begin();
  Wire.setWireTimeout(25000, true);
  if (!wr8(0x6B, 0x00)) return false;   // wakeup
  wr8(0x1A, 0x03);                       // DLPF 44/42 Гц
  wr8(0x1B, 0x08);                       // gyro ±500
  wr8(0x1C, 0x00);                       // accel ±2g
  return true;
}

bool imuRead() {  // вызывается раз в цикл (200 Гц); false = сбой I2C
  Wire.beginTransmission(0x68);
  if (Wire.endTransmission(false) != 0) { imuErr++; Wire.clearWireTimeoutFlag(); return false; }
  Wire.requestFrom(0x68, (uint8_t)14);
  if (Wire.getWireTimeoutFlag()) { imuErr++; Wire.clearWireTimeoutFlag(); return false; }
  // блокирующее чтение 14 байт: accel xyz, temp, gyro xyz (big-endian)
  ...
  float fwd = (axisPick(0, (int8_t)cfg.axf) - fwdBias);   // смещение вычесть
  float up  = (axisPick(0, (int8_t)cfg.axu) - upBias);
  imuGyro   = (axisPick(0, (int8_t)cfg.axg) - gbTilt) / GYR_LSB;
  float aAcc = atan2(fwd, up) * 57.29578f;
  if (bootN < BOOT_CYCLES) { bootA += aAcc; bootN++; if (bootN == BOOT_CYCLES) imuAngle = bootA / BOOT_CYCLES; return true; }
  float al = cfg.alpha;
  imuAngle = al * (imuAngle + imuGyro * CTRL_DT) + (1 - al) * aAcc;
  return true;
}
```

Смещения акселерометра: `aby`-подобные применяются к своей оси до выбора: вычитать `ab` той оси, которую выбрал `axisPick` (нужна отдельная функция, возвращающая ось с вычетом её abx/aby/abz и знаком). `gyroCalib()`: 3 с копить сырые gyro по трём осям (моторы off, робот неподвижен), среднее → cfg.gbx/gby/gbz, печать результата. `rawCmd()`: печатает сырые значения и |a| в g.

### Task 4: motors.ino — TB6612 + энкодеры

**Files:** Create: `firmware/gyrobot_v5/motors.ino`

**Interfaces:** Produces: `void motorsInit(); void drive(int16_t pa, int16_t pb);` (pa/pb: «+» = вперёд, знаки mvA/mvB внутри), `volatile long totA, totB;` (сырые счётчики), `void manDrive(char ch, int16_t pwm);` (команда t, raw без mv/dead), `io` команды.

- [ ] **Шаг 1: код**

```cpp
volatile long totA = 0, totB = 0;

static void isrA() { totA += (digitalRead(PIN_ECA_B) == HIGH) ? 1 : -1; }
static void isrB() { totB += (digitalRead(PIN_ECB_B) == HIGH) ? 1 : -1; }

void motorsInit() {
  pinMode(PIN_AIN1, OUTPUT); pinMode(PIN_AIN2, OUTPUT);
  pinMode(PIN_BIN1, OUTPUT); pinMode(PIN_BIN2, OUTPUT);
  pinMode(PIN_PWMA, OUTPUT); pinMode(PIN_PWMB, OUTPUT);
  pinMode(PIN_STBY, OUTPUT); digitalWrite(PIN_STBY, HIGH);
  pinMode(PIN_ECA_B, INPUT_PULLUP); pinMode(PIN_ECB_B, INPUT_PULLUP);
  attachInterrupt(digitalPinToInterrupt(PIN_ECA_A), isrA, RISING);
  attachInterrupt(digitalPinToInterrupt(PIN_ECB_A), isrB, RISING);
  // ШИМ 3.9 кГц: Timer1 /8 (фаза-коррект 8 бит)
  analogWrite(PIN_PWMA, 0); analogWrite(PIN_PWMB, 0);
  TCCR1B = (TCCR1B & 0xF8) | 0x02;
}

static void chWrite(uint8_t in1, uint8_t in2, uint8_t pwmPin, int16_t v, int16_t mv, float dead, float maxpwm) {
  v *= mv;
  int16_t mag = (v < 0) ? -v : v;
  if (mag > 0 && mag < (int16_t)dead) mag = (int16_t)dead;      // мёртвая зона
  if (mag > (int16_t)maxpwm) mag = (int16_t)maxpwm;
  if (v > 0)      { digitalWrite(in1, HIGH); digitalWrite(in2, LOW); }
  else if (v < 0) { digitalWrite(in1, LOW);  digitalWrite(in2, HIGH); }
  else            { digitalWrite(in1, LOW);  digitalWrite(in2, LOW); }
  analogWrite(pwmPin, mag);
}

void drive(int16_t pa, int16_t pb) {
  chWrite(PIN_AIN1, PIN_AIN2, PIN_PWMA, pa, (int8_t)cfg.mvA, cfg.dead, cfg.maxpwm);
  chWrite(PIN_BIN1, PIN_BIN2, PIN_PWMB, pb, (int8_t)cfg.mvB, cfg.dead, cfg.maxpwm);
}
```

`manDrive` — то же без mv и dead (прямой ШИМ канала), для `t` и прогона. `io <pin> <0|1>` — pinMode+write, только при e 0.

### Task 5: control.ino — два контура + состояние

**Files:** Create: `firmware/gyrobot_v5/control.ino`

**Interfaces:** Consumes: imuAngle/imuGyro, totA/totB, cfg. Produces: `void controlReset();` (сброс интегралов/позиций, при e/fall), `void controlTick(bool fast)` — вызов каждый цикл 200 Гц; внутри на `%SPEED_DIV==0` — медленная часть. Переменные телеметрии: `vFilt, vRef, posEnc, tiltShift, ivi, iAng(нет), pwmA, pwmB, en, fallen`.

- [ ] **Шаг 1: код (полный алгоритм)**

```cpp
// ---------- состояние ----------
bool en = false, fallen = true;
float vFilt = 0, vRef = 0, tiltShift = 0, ivi = 0;
float posEnc = 0, posRef = 0, yawDiff = 0;
int16_t pwmA = 0, pwmB = 0;
float cmdV = 0, cmdTurn = 0;   // от команды d
uint32_t uprightMs = 0;

static long lastTotA = 0, lastTotB = 0;

void controlReset() {   // при выключении, падении, включении заново
  ivi = 0; vFilt = 0; tiltShift = 0; yawDiff = 0;
  lastTotA = totA; lastTotB = totB;
  posRef = posEnc; cmdV = 0; cmdTurn = 0;
}

void controlTick(uint16_t cyc) {
  // --- медленная часть: 50 Гц ---
  if (cyc % SPEED_DIV == 0) {
    noInterrupts();
    long a = totA, b = totB;
    interrupts();
    float da = (a - lastTotA) * cfg.esA, db = (b - lastTotB) * cfg.esB;
    lastTotA = a; lastTotB = b;
    float vRaw = (da + db) * 0.5f;                 // тиков за период
    vFilt = cfg.sfilt * vFilt + (1 - cfg.sfilt) * vRaw;
    float vBody = vFilt - cfg.kth * imuGyro;       // поправка на θ̇ (kth=0 по умолч.)
    posEnc += vRaw;
    // уставка: позиция (возврат на место) + команда водителя
    vRef = -cfg.kpos * (posEnc - posRef) + cmdV;
    // PI → tilt_shift (знак: движется вперёд — наклониться назад)
    float err = vRef - vBody;
    if (err > 20) err = 20; if (err < -20) err = -20;   // кламп ошибки
    ivi += cfg.kii * err;
    if (ivi > 50) ivi = 50; if (ivi < -50) ivi = -50;   // кламп интеграла (°)
    tiltShift = -(cfg.kvi * err + ivi);
    if (tiltShift > cfg.tiltmax) tiltShift = cfg.tiltmax;
    if (tiltShift < -cfg.tiltmax) tiltShift = -cfg.tiltmax;
  }
  // --- быстрая часть: 200 Гц ---
  float setp = cfg.zero + tiltShift;
  float u = cfg.kp * (imuAngle - setp) - cfg.kd * imuGyro;  // + = вперёд
  float diff = cmdTurn;                                      // + = поворот вправо
  pwmA = (int16_t)constrain(u + diff, -255, 255);
  pwmB = (int16_t)constrain(u - diff, -255, 255);
}
```

Машина состояний (в main): fallen = |imuAngle − cfg.zero| > cfg.fall. Если fallen или !en → `drive(0,0)`, сброс `controlReset()` (once), upright-таймер сброшен. Если not fallen и en: если был fallen → после 500 мс непрерывной стойкости включить моторы (arec=1 разрешает). `burst`-захват: буфер `int16_t burst[BURST_N][3]` (угол×100, gyro×10, pwmA), заполняется каждый цикл по команде, выгрузка CSV в protocol.

```cpp
#include "config.h"
Config cfg;
const char* PNAMES[NPARAM] = { "kp","kd","zero","alpha","dead","maxpwm","fall","arec","autoen",
  "kvi","kii","tiltmax","kpos","sfilt","kth","ktp","ktd",
  "esA","esB","mvA","mvB","gbx","gby","gbz","abx","aby","abz","axu","axf","axg","axt" };

uint32_t tCtrl = 0; uint16_t cyc = 0; uint32_t hzCnt = 0, hzVal = 0, hzT = 0;
uint16_t bootN2 = 0;

void setup() {
  Serial.begin(115200);
  loadEEP();
  if (!imuInit()) Serial.println(F("IMU FAIL"));
  motorsInit();
  en = cfg.autoen > 0.5f;
  controlReset();
  Serial.println(F("gyrobot v5"));
}

void loop() {
  uint32_t now = micros();
  if (now - tCtrl >= 5000) {           // 200 Гц
    tCtrl = now;
    imuRead();
    cyc++;
    // hz
    hzCnt++;
    if (now - hzT >= 1000000) { hzVal = hzCnt; hzCnt = 0; hzT = now; }
    // fall/arec state machine + controlTick + drive — по Task 5
    ...
    if (cyc % TELE_DIV == 0) teleLine();
    burstTick();
  }
  serialPoll();                        // разбор команд (неблокирующе)
}
```
